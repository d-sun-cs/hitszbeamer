\section{已完成工作与结果}

\begin{frame}{已完成工作与结果}{Two-tone 如何编码和恢复}

    \setbeamerfont{caption}{size=\scriptsize}
    \begin{columns}[T,onlytextwidth]
        \column{0.38\textwidth}
        \begin{block}{编码时做什么}
            \scriptsize
            \begin{itemize}
                \setlength{\itemsep}{1pt}
                \item 数据被切成 $k$ 个数据块
                \item 每个校验块由数据块移位后异或得到
                \item 移位靠地址偏移完成，核心计算是异或
            \end{itemize}
        \end{block}

        \begin{block}{恢复时做什么}
            \scriptsize
            \begin{itemize}
                \setlength{\itemsep}{1pt}
                \item 先把还活着的数据代入校验块
                \item 按 Two-tone 顺序恢复丢失数据
                \item 如果校验块也丢了，同步补回来
            \end{itemize}
        \end{block}

        \centering

\begin{frame}{已完成工作与结果}{慢在哪里：不是异或慢，而是反复搬数据}

    \begin{columns}[T,onlytextwidth]
        \column{0.42\textwidth}
        \begin{block}{直观实现会做很多重复访问}
            \footnotesize
            \begin{itemize}
                \setlength{\itemsep}{3pt}
                \item 编码时，一个数据块会分散写入多个校验块
                \item 解码时，数据恢复和校验修复被分成两段流程
                \item 同一批校验块会被多轮读写，缓存局部性差
            \end{itemize}
        \end{block}

        \begin{block}{FastTT 的判断}
            \footnotesize
            Two-tone 的异或计算很轻，真正拖慢吞吐的是内存访问次数、访问顺序和缓存命中率。
        \end{block}

\begin{frame}{已完成工作与结果}{FastTT 做了四个具体改动}

    \setbeamerfont{caption}{size=\scriptsize}
    \begin{columns}[T,onlytextwidth]
        \column{0.38\textwidth}
        \begin{block}{对应四个瓶颈}
            \scriptsize
            \begin{itemize}
                \setlength{\itemsep}{1pt}
                \item 校验修复多跑一遍 $\Rightarrow$ 边解码边修校验
                \item 长块跨来跨去访问 $\Rightarrow$ 切成缓存窗口
                \item 校验块写入太分散 $\Rightarrow$ 相邻数据先合并
                \item 单块丢失最常见 $\Rightarrow$ 单遍直接恢复
            \end{itemize}
        \end{block}

        \begin{block}{落地形态}
            \scriptsize
            全部集成进 FastTT，并以 Ceph 插件完成系统验证。
        \end{block}

\begin{frame}{已完成工作与结果}{边恢复数据，边修复丢失校验}

    \setbeamerfont{caption}{size=\scriptsize}
    \begin{columns}[T,onlytextwidth]
        \column{0.36\textwidth}
        \begin{block}{原来的问题}
            \scriptsize
            解码出数据后再单独重编码修复校验块，会多访问一轮数据。
        \end{block}

        \begin{block}{FastTT 的做法}
            \scriptsize
            \begin{itemize}
                \setlength{\itemsep}{1pt}
                \item 把丢失数据块映射到存活校验块
                \item 把丢失校验块映射到自身结构
                \item 每恢复一个码字就同步更新
                \item 解码吞吐提升约 \textbf{100.2\%}
            \end{itemize}
        \end{block}

        \centering

\end{frame}

\begin{frame}{已完成工作与结果}{把长块切成窗口，让缓存真正用起来}

    \setbeamerfont{caption}{size=\scriptsize}
    \begin{columns}[T,onlytextwidth]
        \column{0.38\textwidth}
        \begin{block}{原来的问题}
            \footnotesize
            长块直接遍历时，数据块和校验块之间来回跳转，相关数据还没用完就可能离开缓存。
        \end{block}

        \begin{block}{FastTT 的做法}
            \footnotesize
            \begin{itemize}
                \setlength{\itemsep}{3pt}
                \item 把每个长块划分为连续窗口
                \item 当前窗口内完成移位和异或
                \item 首次访问时再初始化，避免整块预清零
            \end{itemize}
        \end{block}

        \column{0.62\textwidth}
        \centering
        \includegraphics[width=0.98\textwidth,height=0.60\textheight,keepaspectratio]{Cache-Friendly Window效果演示.pdf}

        \vspace{0.1em}
        \scriptsize 同色片段表示同一窗口内的局部访问。
    \end{columns}

\end{frame}

\begin{frame}{已完成工作与结果}{相邻数据先合并，校验块少写一半}

    \setbeamerfont{caption}{size=\scriptsize}
    \begin{columns}[T,onlytextwidth]
        \column{0.34\textwidth}
        \begin{block}{原来的问题}
            \scriptsize
            横向遍历虽然连续读数据，但每个数据块都要分散写入多个校验块，写放大明显。
        \end{block}

        \begin{block}{FastTT 的做法}
            \scriptsize
            相邻两个数据块先在寄存器内异或合并，再一次性写回校验块。
        \end{block}

        \begin{block}{直接结果}
            \scriptsize
            校验块写入访问量由约 $km$ 降至约 $km/2$。
        \end{block}

        \column{0.66\textwidth}
        \centering
        \scriptsize\textbf{合并后：相邻数据合并，再连续写回校验块}

\end{frame}

\begin{frame}{已完成工作与结果}{单块丢失最常见，直接走快路径}

    \setbeamerfont{caption}{size=\scriptsize}
    \begin{columns}[T,onlytextwidth]
        \column{0.38\textwidth}
        \begin{block}{为什么值得单独优化}
            \footnotesize
            \begin{itemize}
                \setlength{\itemsep}{3pt}
                \item 大规模存储中单块失效最常见
                \item 单块丢失不需要完整 Two-tone 消除
                \item 直接对相关存活块做一遍异或即可恢复
            \end{itemize}
        \end{block}

        \begin{block}{直接结果}
            \footnotesize
            \begin{itemize}
                \setlength{\itemsep}{3pt}
                \item 单块丢失相较 ISA-L 提升超过 \textbf{130\%}
                \item 最高提升达到 \textbf{182.1\%}
            \end{itemize}
        \end{block}
