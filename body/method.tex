\begin{frame}{主要研究内容及研究方案}{研究内容 - 存储编码的基本原理}

    % \textbf{存储编码的基本原理}

    \begin{columns}

        \column{0.5\textwidth}

        \begin{block}{RS纠删码编码}
            \footnotesize
            \begin{itemize}
                \item $D$：原始数据块向量
                \item $G$：生成矩阵（Vandermonde/Cauchy）
                \item $C$：编码后的校验块向量
            \end{itemize}
            $$\mathbf{C} = \mathbf{G} \times_{GF[2^w]} \mathbf{D}$$
            原始数据均分成$k$块，通过有限域矩阵乘法编码出$m$块，共$n=k+m$块存储至$n$台机器，至多容忍$m$台故障
        \end{block}

        \column{0.5\textwidth}

        \column{0.5\textwidth}

        \begin{block}{RS纠删码解码}
            \footnotesize
            \begin{itemize}
                \item $D'$：未丢失数据/校验块构成的向量
                \item $G'^{-1}$：由丢失情况构造的逆生成矩阵
                \item $D$：解出的原始数据块向量
            \end{itemize}
            $$\mathbf{D} = \mathbf{G'}^{-1} \times_{GF[2^w]} \mathbf{D'}$$
        \end{block}

    \setbeamerfont{caption}{size=\scriptsize}
    \vspace{-0.8em}
    \begin{columns}

        \column{0.5\textwidth}

        \begin{block}{Two-tone纠删码编码}
            \footnotesize
            \begin{itemize}
                \item $D$：原始数据块向量
                \item $Z$：移位生成矩阵（Two-tone矩阵）
                \item $C$：编码后的校验块向量
            \end{itemize}
            $$\mathbf{C} = \mathbf{Z} \times \mathbf{D}$$
            \vspace{-1.5em}
            \begin{itemize}
                \item $D=\{D_1, D_2, \dots, D_k\}$
                \item $C=\{C_1, C_2, \dots, C_m\}$
                \item $z^{s_{ij}}D_j$：对$D_j$非循环移位$s_{i,j}$个单位
            \end{itemize}
            $$C_i = \bigoplus_{j=1}^{k} z^{s_{ij}}D_j$$
        \end{block}

\begin{frame}{主要研究内容及研究方案}{研究内容 - 优化问题描述}

    \begin{block}{优化问题描述}

        Two-tone纠删码在理论上具备高效性，但工程实现与优化仍面临问题：
        \begin{itemize}
            \item 如何统一处理所有丢失情况，实现解码和重编码的泛化，兼顾高效性
            \item 如何提高对不同偏移量数据块和校验块的访问效率，缓解内存瓶颈
            \item 如何针对只丢失一块的常见场景进一步优化，发挥硬件指令优势
        \end{itemize}

    \end{block}

    基于以上理论和问题，本课题给出了一系列研究方案。

\end{frame}

\subsection{研究方案}

\begin{frame}{主要研究内容及研究方案}{研究方案}

    \setbeamerfont{caption}{size=\scriptsize}

    \small
    \begin{block}{统一解码抽象：Erasure-Parity Mapping}
        \footnotesize
        将所有丢失块的数据/校验块统一抽象为Erasure结构，丢失数据块按序匹配未丢失校验块，丢失校验块匹配自己，统一处理解码与重编码，减少遍历访存次数
    \end{block}

\begin{frame}{主要研究内容及研究方案}{研究方案}

    \begin{columns}

        \column{0.5\textwidth}

        \begin{block}{缓存友好的窗口划分：Cache-Friendly Window}
            \begin{itemize}
                \item 将较长的数据块（chunk）划分为更小的窗口（window）
                \item 在窗口内首次访问时初始化
                \item off-by-one代入策略：领先一个窗口代入以覆盖所有块的偏移
                \item 增强局部性，提高缓存利用率
            \end{itemize}
        \end{block}

    \begin{columns}

        \column{0.45\textwidth}

        \begin{block}{内存友好的相邻合并访问：\\Neighbor-Merging Access}

            \small
            一般的访问模式：
            \begin{itemize}
                \item 横向：每次遍历一整个数据块，对校验块访问反复且不集中
                \item 纵向：按校验块单位访问数据块单位，不同校验块单位偏移不同，对数据块访问不集中
            \end{itemize}  

        \end{block}

\begin{frame}{主要研究内容及研究方案}{研究方案}

    \setbeamerfont{caption}{size=\footnotesize}

    \begin{columns}

        \column{0.5\textwidth}

        \begin{block}{内存友好的相邻合并访问：\\Neighbor-Merging Access}

            \small
            相邻合并访存：结合横向与纵向模式
            \begin{itemize}
                \item 两个相邻数据块内部纵向访存并计算，结果暂存至寄存器
                \item 合并后总体上横向访问校验块
                \item 既加强对数据块访问的集中性，又减少对校验块的重复访问
            \end{itemize}  

        \end{block}

                    \item 总体对所有块只访问一次，可用AVX2的stream指令绕过缓存，进一步提速
