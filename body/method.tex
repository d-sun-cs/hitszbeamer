\subsection{原理瓶颈与优化框架}

\begin{frame}{已完成的主要研究工作}{Two-tone 编解码原理与内存瓶颈}

    \setbeamerfont{caption}{size=\scriptsize}
    \scriptsize
    \begin{columns}[T,onlytextwidth]
        \column{0.40\textwidth}
        \begin{block}{编解码过程}
            \begin{itemize}
                \setlength{\itemsep}{1pt}
                \item 系统化 $(k,k+m)$ 码：前 $k$ 块为数据，后 $m$ 块为校验
                \item 编码：将不同偏移的数据块异或到校验块
                \item 解码：代入存活数据，连续恢复丢失数据并补齐校验块
            \end{itemize}
        \end{block}

        \begin{block}{瓶颈判断}
            \begin{itemize}
                \setlength{\itemsep}{1pt}
                \item 移位靠地址偏移实现，算术开销很低
                \item 直观实现反复读写校验块，瓶颈主要来自\textbf{内存访问}
            \end{itemize}
        \end{block}

\subsection{四项系统级优化}

\begin{frame}{已完成的主要研究工作}{统一解码抽象：同步恢复数据块与校验块}

    \setbeamerfont{caption}{size=\scriptsize}
    \begin{columns}[T]
        \column{0.42\textwidth}
        \begin{block}{核心思想与收益}
            \footnotesize
            \begin{itemize}
                \setlength{\itemsep}{3pt}
                \item 丢失数据块映射到存活校验块
                \item 丢失校验块映射到自身结构
                \item 每解出一个码字立即传播，同步完成数据恢复与校验修复
                \item 省去独立的校验复制与重编码修复阶段
                \item 主要减少内存访问，带来约 \textbf{100.2\%} 的解码提升
            \end{itemize}
        \end{block}

\begin{frame}{已完成的主要研究工作}{缓存友好窗口划分：提升局部性}

    \setbeamerfont{caption}{size=\scriptsize}
    \scriptsize
    \begin{columns}[T,onlytextwidth]
        \column{0.42\textwidth}
        \begin{block}{为什么切窗口}
            \begin{itemize}
                \setlength{\itemsep}{1pt}
                \item 长块直接遍历会频繁跨块访问
                \item 校验块与数据块的错位访问容易破坏缓存局部性
                \item 先处理小窗口，可把相关数据留在缓存附近
            \end{itemize}
        \end{block}

        \begin{block}{实现要点}
            \begin{itemize}
                \setlength{\itemsep}{1pt}
                \item 将每个长块划分为连续窗口
                \item 当前窗口内完成移位异或后再进入下一个窗口
                \item 首次访问时初始化，避免整块预清零
            \end{itemize}
        \end{block}

        \column{0.58\textwidth}
        \centering
        \includegraphics[width=0.98\textwidth,height=0.58\textheight,keepaspectratio]{Cache-Friendly Window效果演示.pdf}

        \vspace{0.2em}
        \scriptsize 同色片段表示同一窗口内的局部访问，窗口间只保留必要重叠。
    \end{columns}

\end{frame}

\begin{frame}{已完成的主要研究工作}{相邻合并访问：降低校验块写放大}

    \setbeamerfont{caption}{size=\scriptsize}
    \begin{columns}[T]
        \column{0.40\textwidth}
        \begin{block}{访问模式对比}
            \footnotesize
            \begin{itemize}
                \setlength{\itemsep}{3pt}
                \item 横向遍历：数据局部性好，但校验块写入分散
                \item 纵向遍历：校验写入集中，但重复读取数据块
                \item 相邻合并：先在寄存器内合并相邻数据，再一次写回校验块
            \end{itemize}
        \end{block}

        \vspace{0.2em}
        \begin{block}{收益}
            \footnotesize
            校验块写入访问量由约 $km$ 降至约 $km/2$，同时保持数据块的连续读取优势。
        \end{block}

\begin{frame}{已完成的主要研究工作}{遍历模式选择：单块丢失快路径}

    \setbeamerfont{caption}{size=\scriptsize}
    \scriptsize
    \begin{columns}[T,onlytextwidth]
        \column{0.42\textwidth}
        \begin{block}{为什么需要特化}
            \begin{itemize}
                \setlength{\itemsep}{1pt}
                \item 大规模存储中单块丢失最常见
                \item 完整 Two-tone 消除在该场景下开销过重
                \item 纵向单遍路径使每个相关码字恰好访问一次
            \end{itemize}
        \end{block}

        \begin{block}{效果}
            \begin{itemize}
                \setlength{\itemsep}{1pt}
                \item 单块丢失相较 ISA-L 提升超过 \textbf{130\%}
                \item 最高提升达到 \textbf{182.1\%}
                \item 多块丢失自动回退到统一解码与标准消除流程
            \end{itemize}
        \end{block}
