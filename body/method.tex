\begin{frame}{已完成工作与结果}{FastTT 做了四个具体改动}

    \setbeamerfont{caption}{size=\scriptsize}
    \begin{columns}[T,onlytextwidth]
        \column{0.38\textwidth}
        \begin{block}{四个改动}
            \scriptsize
            \begin{enumerate}
                \setlength{\itemsep}{3pt}
                \item 解出的数据块立刻去修校验块
                \item 长块切成短块窗口（Tiling）
                \item 相邻块先在寄存器合并再访存
                \item 单块丢失走快速解码路径
            \end{enumerate}
        \end{block}

        \begin{block}{落地结果}
            \scriptsize
            四项优化全部进入 FastTT，并通过 Ceph 插件验证。
        \end{block}

        \column{0.62\textwidth}
        \centering
        \includegraphics[width=0.98\textwidth,height=0.6\textheight,keepaspectratio]{Two-tone Optimization Overview.pdf}

        \vspace{0.15em}
        \scriptsize 编码侧优化写入路径；\\ 解码侧同时减少冗余流程和常见故障开销。
    \end{columns}

\end{frame}

\begin{frame}{已完成工作与结果}{解出的数据块立刻去修校验块（Erasure-Parity Mapping）}

    \setbeamerfont{caption}{size=\scriptsize}
    \begin{columns}[T,onlytextwidth]
        \column{0.38\textwidth}
        \begin{block}{原来的问题}
            \scriptsize
            先解码修复数据块，再重新编码修复校验块：多跑了一遍。
        \end{block}

        \begin{block}{FastTT 的做法}
            \scriptsize
            \begin{itemize}
                \setlength{\itemsep}{3pt}
                \item 丢失数据块映射到存活校验块
                \item 丢失校验块映射到自身
                \item 图例：$d_1$、$d_2$ 解出码字后立刻用于重编码 $c_2$
            \end{itemize}
        \end{block}

        \column{0.62\textwidth}
        \centering
        \includegraphics[width=0.98\textwidth,height=0.55\textheight,keepaspectratio]{Erasure-Parity Mapping Decoding Demonstration.pdf}
    \end{columns}

\end{frame}

\begin{frame}{已完成工作与结果}{长块切成短块窗口（Cache-Friendly Window Partitioning）}

    \setbeamerfont{caption}{size=\scriptsize}
    \begin{columns}[T,onlytextwidth]
        \column{0.38\textwidth}
        \begin{block}{原来的问题}
            \scriptsize
            对校验块来回跳转访问，数据还没用完就离开缓存。
        \end{block}

        \begin{block}{FastTT 的做法}
            \scriptsize
            \begin{itemize}
                \setlength{\itemsep}{3pt}
                \item 长块切成短块窗口
                \item 窗口内完成移位和异或
                \item 图例：同色片段在同一窗口内访问，编码解码都应用
            \end{itemize}
        \end{block}

        \column{0.62\textwidth}
        \centering

\end{frame}

\begin{frame}{已完成工作与结果}{相邻块先在寄存器合并再访存（Neighbor-Merging Access）}

    \setbeamerfont{caption}{size=\tiny}
    \vspace{-0.25em}
    \begin{columns}[T,onlytextwidth]
        \column{0.30\textwidth}
        \begin{block}{原来的问题}
            \scriptsize
            横向模式重复写校验块内存，纵向模式重复读数据块内存。
        \end{block}

        \begin{block}{FastTT 的做法}
            \scriptsize
            \begin{itemize}
                \setlength{\itemsep}{2pt}
                \item 数据块码字异或结果先存到向量寄存器，再连续写回校验块内存
                \item 注：具体实现中一个码字256位，等于AVX2向量宽度
                \item 写入校验块内存访问减少约一半，对数据块内存也没有重复读取
            \end{itemize}
        \end{block}

        \begin{block}{原来的问题}
            \scriptsize

        \begin{block}{FastTT 的做法}
            \scriptsize
            \begin{itemize}
                \setlength{\itemsep}{2pt}
                \item 单块丢失选择纵向遍历
                \item 存活块异或一遍即可恢复
                \item 图例：缺失块一次异或得到
            \end{itemize}
        \end{block}
