\begin{frame}{主要研究内容及研究方案}{研究内容}

    \textbf{存储编码的基本原理}

    \begin{columns}

        \column{0.5\textwidth}

        \begin{block}{RS纠删码编码}
            \footnotesize
            \begin{itemize}
                \item $C$：原始数据块向量
                \item $G$：生成矩阵（Vandermonde/Cauchy）
                \item $C$：编码后的校验块向量
            \end{itemize}
            $$\mathbf{C} = \mathbf{G} \times_{GF[2^w]} \mathbf{D}$$
            原始数据均分成$k$块，通过有限域矩阵乘法编码出$m$块，共$n=k+m$块存储至$n$台机器，至多容忍$m$台故障
        \end{block}

        \column{0.5\textwidth}

    \textbf{存储编码的基本原理}

    \begin{columns}

        \column{0.5\textwidth}

        \begin{block}{RS纠删码解码}
            \footnotesize
            \begin{itemize}
                \item $D'$：未丢失数据/校验块构成的向量
                \item $G'^{-1}$：由丢失情况构造的逆生成矩阵
                \item $D$：解出的原始数据块向量
            \end{itemize}
            $$\mathbf{D} = \mathbf{G'}^{-1} \times_{GF[2^w]} \mathbf{D'}$$
            原始数据均分成$k$块，通过有限域矩阵乘法编码出$m$块，共$n=k+m$块存储至$n$台机器，至多容忍$m$台故障
        \end{block}

        \column{0.5\textwidth}

\subsection{研究方案}
\begin{frame}{研究方案}
    \begin{block}{课题主要内容}
        研究Two-tone移位异或纠删码的高效实现，将其以插件形式集成至Ceph分布式存储系统，提出系统级优化方法，提升编解码性能。
    \end{block}
\end{frame}

\begin{frame}{面临的挑战}
    \begin{block}{面临的挑战}
        \begin{itemize}
            \item 如何统一处理所有丢失情况，实现解码和重编码的泛化。
            \item 如何提升缓存局部性，减少内存访问压力。
            \item 如何针对单块丢失场景特化优化。
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{拟采用的方法}
    \begin{block}{拟采用的方法}
        \begin{itemize}
            \item 统一解码抽象（Erasure-Parity Mapping）
            \item 缓存友好窗口划分（Cache-Friendly Window）
            \item 内存友好相邻合并访问（Neighbor-Merging Access）
            \item 特化优化（Special Case Expansibility）
        \end{itemize}
    \end{block}
\end{frame}
