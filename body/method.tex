\subsection{原理瓶颈与优化框架}

\begin{frame}{已完成的主要研究工作}{Two-tone编解码原理与性能瓶颈}

    \setbeamerfont{caption}{size=\scriptsize}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \footnotesize
        \textbf{编码}：$c_{i}[x]=\bigoplus_{j=1}^{k} d_j[x-s_{i,j}]$，移位由地址偏移实现，几乎无开销。\\[0.3em]
        \textbf{解码}（连续可解三阶段）：代入 $\to$ 消除 $\to$ 修复。
        \vspace{0.3em}
        \begin{block}{瓶颈：内存受限，而非算术}
            \footnotesize
            \begin{itemize}
                \item 冗余访存、时间局部性差
                \item 横向/纵向遍历各有访存瓶颈
                \item 解码与修复分离、统一遍历致次优
            \end{itemize}
        \end{block}

        \column{0.52\textwidth}
        \begin{figure}
            \centering
            \includegraphics[width=\textwidth,height=0.66\textheight,keepaspectratio]{Example of Two-tone Decoding.pdf}
            \vspace{-0.6em}
            \caption{Two-tone解码举例（$\boldsymbol d_1,\boldsymbol d_2,\boldsymbol c_2$丢失）}
        \end{figure}
    \end{columns}

\end{frame}

\subsection{四项优化与系统集成}

\begin{frame}{已完成的主要研究工作}{优化一、二：EPM 与 CFWP}

    \footnotesize
    \begin{columns}[T]
        \column{0.5\textwidth}
        \textbf{EPM：统一解码抽象}
        \begin{itemize}
            \item 丢失\emph{数据块}$\to$存活校验块；丢失\emph{校验块}$\to$自身
            \item 修复融入消除，3趟$\Rightarrow$2趟，省去独立修复
        \end{itemize}
        \centering
        \includegraphics[width=0.82\textwidth,height=0.36\textheight,keepaspectratio]{Erasure-Parity Mapping Decoding Demonstration.pdf}

        \column{0.5\textwidth}
        \textbf{CFWP：缓存友好窗口划分}
        \begin{itemize}
            \item 数据块切分为定长\textbf{窗口}，窗口内完成全部移位异或，提升缓存命中
            \item 首次访问时初始化；解码 off-by-one 领先一窗口；$W$取$\ge k{-}1$的最小2的幂
        \end{itemize}
        \centering
        \includegraphics[width=0.92\textwidth,height=0.36\textheight,keepaspectratio]{Cache-Friendly Window Demonstration.pdf}
    \end{columns}

\end{frame}

\begin{frame}{已完成的主要研究工作}{优化三、四：NMA 与 TPS}

    \footnotesize
    \begin{columns}[T]
        \column{0.5\textwidth}
        \textbf{NMA：相邻合并访问}
        \begin{itemize}
            \item 相邻两数据块码字先在SIMD寄存器内异或，再一次写回校验块
            \item 校验块写入理论减半 $\Theta((k{+}\tfrac{km}{2})l)$，优于横向/纵向
        \end{itemize}
        \centering
        \includegraphics[width=0.92\textwidth,height=0.38\textheight,keepaspectratio]{Neighbor-Merging Access Pattern.pdf}

        \column{0.5\textwidth}
        \textbf{TPS：遍历模式选择}
        \begin{itemize}
            \item 针对最常见\textbf{单块丢失}：免代入与消除，一遍纵向全局异或恢复
            \item 每码字恰访问一次，用AVX2 stream指令绕过缓存
        \end{itemize}
        \centering
        \includegraphics[width=0.82\textwidth,height=0.38\textheight,keepaspectratio]{Vertical Traversal Pattern Demonstration.pdf}
    \end{columns}

\end{frame}
