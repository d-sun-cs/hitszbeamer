\begin{frame}{后期工作、困难与可行性}{困难、解决方案与可行性}

    \footnotesize
    \begin{columns}[T]
        \column{0.56\textwidth}
        \begin{block}{主要困难与应对}
            \footnotesize
            \begin{itemize}
                \setlength{\itemsep}{3pt}
                \item $(10,4)$等配置受 AVX2 寄存器约束影响\quad $\to$ 更灵活的码字分组、SIMD 调度或 AVX-512 探索
                \item 两块丢失场景提升有限\quad $\to$ 设计介于单块快路径与通用消除间的中间策略
                \item 方法论泛化仍需论证\quad $\to$ 抽象与具体偏移结构无关的访存优化原则
                \item KV Cache 方向存在收益不确定性\quad $\to$ 先构建小型原型，评估容量收益与恢复延迟权衡
            \end{itemize}
        \end{block}

            \vspace{0.6em}
            \begin{center}
                \large\textbf{能够按期完成全部论文工作。}
            \end{center}
        \end{block}
