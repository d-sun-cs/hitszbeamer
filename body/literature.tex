    \begin{block}{国内外研究现状}
        纠删码作为分布式存储系统的核心技术，传统RS码\cite{reedPolynomialCodesCertain1960}具有良好的MDS特性。为提升效率，研究者提出了异或优化\cite{lubyXORBasedErasureResilientCoding1995}、矩阵结构优化\cite{wangFastAccelerationStrategies2025}等方法。近年来，移位异或码如ZigZag\cite{gongZigzagDecodableCodes2017}和Two-tone\cite{fuTwotoneShiftXORStorage2021}成为新热点，具备更高的编解码性能。
    \end{block}
    \begin{block}{主要参考文献}
        \begin{itemize}
            \item Reed和Solomon（1960）提出了RS码\cite{reedPolynomialCodesCertain1960}。
            \item Luby等（1995）提出异或优化\cite{lubyXORBasedErasureResilientCoding1995}。
            \item Gong等（2017）提出ZigZag码\cite{gongZigzagDecodableCodes2017}。
            \item Fu等（2021）提出Two-tone码\cite{fuTwotoneShiftXORStorage2021}。
            \item Wang等（2025）提出Cerasure系统优化\cite{wangFastAccelerationStrategies2025}。
        \end{itemize}
    \end{block}
