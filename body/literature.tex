\begin{frame}{后期工作、困难与可行性}{新增创新点：面向 KV Cache 存储的移位异或码}

    \footnotesize
    \begin{columns}[T]
        \column{0.5\textwidth}
        \begin{block}{问题与机会}
            \footnotesize
            LLM推理KV Cache规模巨大（如LLaMA3-70B@128k约40\,GB），分布式缓存池（Mooncake）多用多副本、利用率低；纠删码冗余更省，但重建在\textbf{延迟关键路径}上，RS码开销大——移位异或码仅靠移位异或，配合FastTT高吞吐恰好可行。
        \end{block}
        \begin{block}{研究内容与收益}
            \footnotesize
            把纠删码定位为\textbf{“性能导向的冗余”}：替代多副本提升等效容量与命中率，故障时快速就地重建KV块、避免GPU重算，维持TTFT等SLO。
        \end{block}

        \column{0.5\textwidth}
        \begin{block}{进度安排}
            \footnotesize
            \begin{itemize}
                \setlength{\itemsep}{3pt}
                \item \textbf{2026.03--04}：完善论文与补充实验，凝练方法论，规划创新点
                \item \textbf{2026.04--05}：新创新点初步研究与验证，扩展实验，整理数据
                \item \textbf{2026.05--06}：撰写毕业论文，完善图表，准备答辩
            \end{itemize}
        \end{block}
    \end{columns}

\end{frame}

\begin{frame}{后期工作、困难与可行性}{困难、解决方案与可行性}

    \footnotesize
    \begin{columns}[T]
        \column{0.5\textwidth}
        \textbf{困难 → 解决方案}
        \begin{itemize}
            \setlength{\itemsep}{3pt}
            \item $(10,4)$等配置寄存器约束瓶颈 → 更灵活的码字分组/SIMD调度或AVX-512
            \item 两块丢失优化空间有限 → 设计介于快路径与通用消除间的中间策略
            \item 方法论通用性 → 抽象与偏移结构无关的共性，构建移位异或码族通用框架
            \item 工程集成复杂 → 借助官方工具与单元/端到端测试保证正确性
        \end{itemize}
