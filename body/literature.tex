\section{后期主要工作}

\begin{frame}[t]{后期主要工作}{补足已有结果，再扩展小覆盖写增量更新}

    \vspace{-0.4em}
    \begin{block}{1. 给 FastTT 优化补直接证据}
        \footnotesize
        \begin{itemize}
            \setlength{\itemsep}{2pt}
            \item 在已有吞吐和消融结果外，补测缓存命中率、实际访存次数。
            \item 目标：把“为什么省内存开销”讲清楚，支撑论文整理。
        \end{itemize}
    \end{block}

    \vspace{0.35em}

    \begin{block}{2. 小规模覆盖写场景下的增量更新}
        \footnotesize
        \begin{itemize}
            \setlength{\itemsep}{3pt}
            \item （新场景）已有对象只改一小段：计算变化量$\Delta$，只更新受影响数据段和校验段。
            \item 挑战一：连续写入要映射到移位后的校验区间，需解决边界和内存对齐问题。
            \item 挑战二：增量更新不一定比完整重新编码快，需要选择策略。
            \item 挑战三：RS码也能通过有限域运算更新，要探索如何定位性能瓶颈并取得优势。
        \end{itemize}

        \vspace{0.2em}
        \centering
        \scriptsize 产出：局部增量更新模型、性能瓶颈分析、实验结果验证。
    \end{block}

\end{frame}
