\section{研究内容及进度}

\begin{frame}{研究内容及进度}{研究背景与问题定位}

    \scriptsize
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.49\textwidth}
            \begin{block}{分布式存储的容错需求}
                \begin{itemize}
                    \setlength{\itemsep}{1pt}
                    \item 多副本可靠但空间开销高；Reed-Solomon 码高效但有限域运算较重
                    \item Shift-XOR 码仅需移位与异或，具备更高理论性能潜力
                \end{itemize}
            \end{block}

            \begin{block}{Two-tone 的工程缺口}
                \begin{itemize}
                    \setlength{\itemsep}{1pt}
                    \item 缺少覆盖数据块/校验块丢失组合的高性能实现
                    \item 缺少面向缓存局部性、访存路径和常见故障模式的系统优化验证
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.51\textwidth}
            \begin{block}{本课题目标}
                面向 Two-tone 移位异或码构建高性能实现 \textbf{FastTT}，把理论上的移位异或优势转化为可部署的系统吞吐优势。
            \end{block}

            \begin{block}{中期阶段关注点}
                \begin{itemize}
                    \setlength{\itemsep}{1pt}
                    \item 从编解码流程中定位内存访问瓶颈
                    \item 围绕访存次数、缓存局部性和单块丢失快路径设计优化
                    \item 在 Ceph 插件和微基准中验证吞吐收益
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

\end{frame}
