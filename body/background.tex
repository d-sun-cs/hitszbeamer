\section{研究内容及进度}

\begin{frame}{研究内容及进度}{研究背景与问题}

    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \textbf{背景}
            \footnotesize
            \begin{itemize}
                \item 纠删码以远低于多副本的存储开销提供同等可靠性，是GFS、Azure、HDFS、Ceph的核心容错技术
                \item RS码\cite{reedPolynomialCodesCertain1960}基于有限域矩阵乘法，已有大量系统级优化（Jerasure、ISA-L、Cerasure）
                \item Two-tone移位异或码\cite{fuTwotoneShiftXORStorage2021}仅靠移位与异或，\textbf{理论性能优于RS码}，是移位异或码SOTA
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{问题：理论优越但缺乏高效实现}
            \footnotesize
            \begin{itemize}
                \item \textbf{缺少泛化性实现}：无法统一应对所有数据块/校验块丢失
                \item \textbf{缺少系统级优化}：RS码优化因算法本质不同无法直接迁移
                \item \textbf{缺少工业级应用}：尚未集成至Ceph等开源存储系统
            \end{itemize}
            \vspace{0.3em}
            \begin{block}{本课题}
                \footnotesize 以Two-tone移位异或码为对象，研究其\textbf{系统级高效实现}方法。
            \end{block}
        \end{column}
    \end{columns}

\end{frame}
