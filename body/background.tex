\section{研究内容与进度}

\begin{frame}{研究内容与进度}{存储系统需要可靠且低开销的冗余}

    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.47\textwidth}
            \begin{block}{多副本：简单但占空间}
                \footnotesize
                \begin{itemize}
                    \setlength{\itemsep}{4pt}
                    \item 数据需要冗余，才能在节点或磁盘失效后恢复
                    \item 多副本直接保存多份数据，可靠但存储开销高
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.47\textwidth}
            \begin{block}{纠删码：用计算换空间}
                \footnotesize
                \begin{itemize}
                    \setlength{\itemsep}{4pt}
                    \item 只保存少量校验块，也能恢复丢失数据
                    \item 代价是写入、恢复和降级读都需要编解码计算
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

    \vspace{0.7em}
    \begin{block}{本课题的优化重点}
        \centering
        \footnotesize
        让纠删码的编码、解码尽可能快，降低节省空间带来的计算代价。
    \end{block}

\end{frame}

\begin{frame}{研究内容与进度}{为什么选择移位异或纠删码}

    \setbeamerfont{caption}{size=\scriptsize}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.39\textwidth}
        \begin{block}{Reed-Solomon 纠删码}
            \scriptsize
            \begin{itemize}
                \setlength{\itemsep}{2pt}
                \item 可靠、成熟，工程系统广泛使用
                \item 编解码需要有限域矩阵运算
            \end{itemize}
        \end{block}

        \begin{block}{移位异或纠删码}
            \scriptsize
            \begin{itemize}
                \setlength{\itemsep}{2pt}
                \item 移位异或计算，复杂度优于 RS
                \item Two-tone 这一类中的 SOTA
                \item 缺少面向存储系统的高速实现
                \item RS 优化不能直接套用
            \end{itemize}
        \end{block}

        \vspace{0.2em}
        \centering
        \footnotesize\textbf{因此：对 Two-tone 做系统级优化}
        \end{column}

\end{frame}
