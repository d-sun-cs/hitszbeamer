\section{研究内容及进度}

\begin{frame}{研究内容及进度}{研究背景与问题}

    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \begin{block}{为什么需要纠删码}
                \footnotesize
                \begin{itemize}
                    \setlength{\itemsep}{3pt}
                    \item 分布式存储需要在节点/磁盘失效下保持数据可靠
                    \item 多副本实现简单，但存储开销高；纠删码以更低冗余提供同等可靠性
                    \item RS码\cite{reedPolynomialCodesCertain1960}成熟且广泛部署，但依赖有限域运算
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{block}{为什么关注 Two-tone}
                \footnotesize
                \begin{itemize}
                    \setlength{\itemsep}{3pt}
                    \item Two-tone移位异或码\cite{fuTwotoneShiftXORStorage2021}仅由移位和异或构成，理论性能潜力高
                    \item 现有工作缺少覆盖全部丢失组合的泛化实现
                    \item 尚缺少面向缓存、访存和工业系统的系统级优化
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

    \vspace{0.6em}
    \begin{block}{本课题定位}
        \centering
        \footnotesize 面向 Two-tone 码构建高效实现 FastTT：把理论上的移位异或优势转化为可部署的系统吞吐优势。
    \end{block}

\end{frame}
