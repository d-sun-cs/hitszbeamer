\subsection{课题来源}

\begin{frame}{课题的来源及研究目的和意义}{课题来源}

    \begin{columns}[T]

        \begin{column}{0.5\textwidth}

            \textbf{存储系统的数据可靠性}

            \vspace{1em}

            多副本：实现简单、存储开销翻倍

            纠删码：
            \begin{itemize}
                \item 计算复杂，存储开销更小
                \item 可靠性更高\cite{weatherspoonErasureCodingVs2002}
            \end{itemize}

            Google FS、Win Azure和Red Hat开源Ceph等存储系统均支持纠删码

        \end{column}
        \begin{column}{0.5\textwidth}

            \textbf{经典纠删码}

            \vspace{1em}

            Reed-Solomon码（RS码）\cite{reedPolynomialCodesCertain1960}：
            \begin{itemize}
                \item 基于有限域$GF[2^w]$矩阵乘法
                \item 可部分转换为异或运算\cite{lubyXORBasedErasureResilientCoding1995}
            \end{itemize}

            Jerasure\cite{plankJerasureLibraryFacilitating2008}、Intel ISA\cite{gajalwarExploringDataProtection2024}等RS码的优化库在工业级存储系统中应用广泛

        \end{column}

    \end{columns}

\end{frame}

            \begin{itemize}
                \item 基于非循环移位和异或运算

                \item 暂无实现层面的优化研究
            \end{itemize}

\subsection{研究的目的和意义}

\begin{frame}{课题的来源及研究目的和意义}{研究的目的和意义}

    \begin{block}{研究目的}
        \begin{itemize}
            \item 集成Two-tone纠删码至开源分布式存储系统Ceph，完善其功能
            \item 尝试各种系统级优化方案，取得更高的纠删码吞吐性能
        \end{itemize}
    
    \end{block}

    \begin{block}{研究意义}
        \begin{itemize}
            \item 推动移位异或纠删码从算法理论到实际应用，实现更高效的纠删码
            \item 填补位异或纠删码缺少系统级优化的空白，对齐RS码的优化研究
        \end{itemize}
    \end{block}

\end{frame}
