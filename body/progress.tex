\begin{frame}{实验评估与结果}{与 Cerasure 的学术基线对比}

        \column{0.32\textwidth}
        \begin{block}{评价方式}

        \end{block}

            \begin{itemize}
                \setlength{\itemsep}{1pt}

\begin{frame}{实验评估与结果}{消融实验：四项优化贡献}

        \begin{block}{从原始实现到完整实现}
            \footnotesize
            \begin{itemize}
                \setlength{\itemsep}{3pt}
                \item 统一解码消除冗余修复访存
                \item 缓存友好窗口提升局部性
