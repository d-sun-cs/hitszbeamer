\section{预期目标与进度安排}
\subsection{预期目标}

\begin{frame}{预期目标与进度安排}{预期目标}

  \begin{block}{系统集成与应用目标}

  \small
  \begin{itemize}
    \setlength{\itemsep}{1pt}
    \item Two-tone纠删码插件化集成至Ceph分布式存储系统
    \item 接口集成与功能验证：编码/解码正确性与性能测试
    \item 集群集成与应用验证：在测试集群下的读写流程与稳定性
  \end{itemize}

  \end{block}

  \begin{block}{性能优化与总结方法论目标}

  \footnotesize
  \begin{itemize}
    \setlength{\itemsep}{1pt}
    \item 实现提出的优化研究方案，总结移位异或纠删码系统级优化方法论
    \vspace{-0.5em}
    \begin{itemize}
      \item 统一解码抽象：提升解码泛化能力与效率
      \item 缓存友好窗口划分：优化数据块与校验块缓存局部性
      \item 内存友好相邻合并访问：减少重复访问，提升编码效率
      \item 特化优化：只丢失一块的场景下高效解码
    \end{itemize}
    \item 与Ceph中的SOTA纠删码ISA和学术研究SOTA纠删码Cerasure\cite{wangFastAccelerationStrategies2025}对比性能
  \end{itemize}

  \end{block}

\end{frame}
