\section{实验评估与结果}

\begin{frame}{实验评估与结果}{Ceph集成与实验设置}

    \footnotesize
    \begin{columns}[T]
        \column{0.5\textwidth}
        \begin{block}{系统集成}
            \footnotesize
            \begin{itemize}
                \setlength{\itemsep}{3pt}
                \item FastTT 以插件形式集成至 \textbf{Ceph v18.2.7}
                \item 通过标准纠删码接口测试，验证编解码正确性
                \item 使用 \texttt{vstart}+\texttt{rados} 完成端到端读写验证
            \end{itemize}
        \end{block}
        \begin{block}{评估目标}
            \footnotesize 不只看微基准吞吐，也验证优化在真实存储系统中的可用性。
        \end{block}

        \column{0.5\textwidth}
        \begin{block}{实验平台与基线}
            \footnotesize
            \begin{itemize}
                \setlength{\itemsep}{3pt}
                \item CPU：i9-14900K，AVX2；内存：64\,GB DDR5
                \item 块大小：2\,MB；多组 $(k,m)$ 配置，重复运行取平均
                \item 基线：Jerasure、\textbf{ISA-L}（Ceph生产级）、\textbf{Cerasure}（学术SOTA）
            \end{itemize}
        \end{block}
        \begin{block}{评价维度}
            \footnotesize 编码吞吐、解码吞吐、优化消融、Ceph端到端 \emph{PUT} 吞吐。
        \end{block}
    \end{columns}

\end{frame}
